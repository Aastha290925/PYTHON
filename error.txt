1. Basic Try-Except Block
try:
    result = 10 / 0
except ZeroDivisionError:
    print("Error: You cannot divide by zero!")

output:
Error: You cannot divide by zero!

2. Handling Multiple Specific Exceptions

import json

test_cases = ["not numbers", "0"]

for case in test_cases:
    try:
        if case == "not numbers":
            result = "10" + 5  # TypeError
        else:
            result = 10 / int(case)  # ZeroDivisionError
    except ZeroDivisionError:
        print("Caught a ZeroDivisionError!")
    except TypeError:
        print("Caught a TypeError!")

output:
Caught a TypeError!
Caught a ZeroDivisionError!

4. Using else and finally

def divide(a, b):
    try:
        print(f"\nAttempting: {a} / {b}")
        result = a / b
    except ZeroDivisionError:
        print("Exception caught: Cannot divide by zero.")
    else:
        print(f"Success! The result is {result}")
    finally:
        print("Cleanup: Execution of divide function finished.")

divide(10, 2)
divide(10, 0)

output:
Attempting: 10 / 2
Success! The result is 5.0
Cleanup: Execution of divide function finished.

Attempting: 10 / 0
Exception caught: Cannot divide by zero.
Cleanup: Execution of divide function finished.

5. Raising Exceptions Manually (raise)

def check_age(age):
    if age < 0:
        raise ValueError("Age cannot be negative!")
    return f"Age {age} is valid."

try:
    check_age(-5)
except ValueError as e:
    print(f"Validation failed: {e}")

output:
Validation failed: Age cannot be negative!

6. Custom/User-defined Exceptions

class InsufficientFundsError(Exception):
    """Exception raised when a bank withdrawal exceeds the available balance."""
    def __init__(self, balance, amount):
        super().__init__(f"Attempted to withdraw ${amount} but balance is only ${balance}.")

balance = 50
withdrawal = 100

try:
    if withdrawal > balance:
        raise InsufficientFundsError(balance, withdrawal)
except InsufficientFundsError as e:
    print(f"Transaction Denied: {e}")

output:
Transaction Denied: Attempted to withdraw $100 but balance is only $50.

7. Exception Chaining (raise ... from)

def get_system_config():
    try:
        # Simulating a core system lookup failure
        raise KeyError("database_url missing")
    except KeyError as original_error:
        raise RuntimeError("Application failed to initialize") from original_error

try:
    get_system_config()
except RuntimeError as e:
    print(f"Caught: {e}")
    print(f"Original root cause: {e.__cause__}")

output:
Caught: Application failed to initialize
Original root cause: 'database_url missing'

================================================
8. Handling Multiple Errors Together (ExceptionGroup)

def fetch_data():
    # Bundling multiple errors together
    raise ExceptionGroup(
        "Network Tasks Failed",
        [ValueError("Invalid URL format"), ConnectionError("Server unreachable")]
    )

try:
    fetch_data()
except* ValueError as eg:
    print("Handled the value errors within the group.")
except* ConnectionError as eg:
    print("Handled the connection errors within the group.")

output:
Handled the value errors within the group.
Handled the connection errors within the group.
