2. Encapsulation (Access Modifiers)

class BankAccount:
    def __init__(self, owner, balance):
        self.owner = owner          # Public
        self._account_type = "Checkings"  # Protected
        self.__balance = balance    # Private 

    def get_balance(self):
        return self.__balance

    def deposit(self, amount):
        if amount > 0:
            self.__balance += amount
            print(f"Deposited ${amount}")
        else:
            print("Invalid deposit amount!")

account = BankAccount("Alice", 1000)
account.deposit(500)
print(f"Account Balance: ${account.get_balance()}")

try:
    print(account.__balance)
except AttributeError as e:
    print(f"Error: {e}")

output:
Deposited $500
Account Balance: $1500
Error: 'BankAccount' object has no attribute '__balance'

A. Single Inheritance

C. Multilevel Inheritance

class Grandfather:
    def land(self):
        return "Grandfather owns a plot of land."

class Father(Grandfather):

print(obj.car())

Grandfather owns a plot of land.

4. Polymorphism (Method Overriding & Duck Typing)

Polymorphism allows different classes to have methods with the same name but different behaviors.

A. Method Overriding

python
class Shape:
    def area(self):
        return "Area not defined."

class Circle(Shape):
    def __init__(self, radius):
        self.radius = radius
    def area(self):  
        return 3.14 * (self.radius ** 2)

class Square(Shape):
    def __init__(self, side):
        self.side = side
    def area(self):  
        return self.side * self.side

shapes = [Circle(5), Square(4)]
for shape in shapes:
    print(f"Calculated Area: {shape.area()}")
Use code with caution.
Output:
text
Calculated Area: 78.5
Calculated Area: 16
Use code with caution.

5. The super() Function

The super() function allows a child class to call the constructor or methods of its parent class, preventing code duplication.
python
class Employee:
    def __init__(self, name, salary):
        self.name = name
        self.salary = salary

class Manager(Employee):
    def __init__(self, name, salary, department):
        # Calls the parent constructor to handle name and salary
        super().__init__(name, salary)
        self.department = department

    def get_details(self):
        return f"Manager: {self.name}, Dept: {self.department}, Salary: ${self.salary}"

mgr = Manager("Bob", 95000, "IT")
print(mgr.get_details())
Use code with caution.
Output:
text
Manager: Bob, Dept: IT, Salary: $95000
Use code with caution.

6. Class Attributes vs. Instance Attributes

• Instance Attribute: Unique to each individual object.
• Class Attribute: Shared equally across all objects of that class.
python
class Student:
    school_name = "Global High School"  # Class Attribute

    def __init__(self, name):
        self.name = name  # Instance Attribute

student1 = Student("Alex")
student2 = Student("Emma")

print(f"{student1.name} goes to {student1.school_name}")
print(f"{student2.name} goes to {student2.school_name}")
Use code with caution.
Output:
text
Alex goes to Global High School
Emma goes to Global High School
Use code with caution.

7. Class Methods and Static Methods

class Calculator:
    brand = "Casio"

    @classmethod
    def get_brand(cls):
        return f"This calculator brand is {cls.brand}."

    @staticmethod
    def add(x, y):
        return x + y

# Called directly on the Class without creating an object
print(Calculator.get_brand())
print(f"Static Add Result: {Calculator.add(15, 30)}")

output:
This calculator brand is Casio.
Static Add Result: 45

8. Abstraction 

from abc import ABC, abstractmethod

class Vehicle(ABC):  # Abstract Base Class
    @abstractmethod
    def start_engine(self):
        pass

class Bike(Vehicle):
    def start_engine(self):
        return "Kickstart or press ignition button to start Bike."

# trying to instantiate Vehicle directly throws an error: Vehicle() -> TypeError
my_bike = Bike()
print(my_bike.start_engine())

output:
Kickstart or press ignition button to start Bike.
