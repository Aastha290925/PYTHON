search_item = input("Enter element to search in t4: ")
found = False

for i in range(len(t4)):
    if t4[i] == search_item:
        print(f"Found '{search_item}' at main index {i}")
        found = True
    elif isinstance(t4[i], tuple) and search_item in t4[i]:
        sub_index = t4[i].index(search_item)
        print(f"Found '{search_item}' at nested index t4[{i}][{sub_index}]")
        found = True

if not found:
    print("Not found")
